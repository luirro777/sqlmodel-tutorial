from sqlmodel import Field, Relationship, SQLModel


# ============================================================
# MODELO "UNO"
# ============================================================

class Team(SQLModel, table=True):

    # Clave primaria de la tabla team
    id: int | None = Field(
        default=None,
        primary_key=True
    )

    name: str

    # Un Team puede tener muchos Hero
    #
    # "Hero" está entre comillas porque la clase Hero
    # todavía no fue declarada en este punto.
    heroes: list["Hero"] = Relationship(
        back_populates="team"
    )


# ============================================================
# MODELO "MUCHOS"
# ============================================================

class Hero(SQLModel, table=True):

    id: int | None = Field(
        default=None,
        primary_key=True
    )

    name: str

    # --------------------------------------------------------
    # FOREIGN KEY
    # --------------------------------------------------------
    #
    # Esta es una columna REAL de la tabla hero.
    #
    # Contendrá el ID del Team al que pertenece el Hero.
    #
    # Por ejemplo:
    #
    # hero.team_id == 1
    #
    # significa que el Hero pertenece al Team cuyo id == 1.
    team_id: int | None = Field(
        default=None,
        foreign_key="team.id"
    )

    # --------------------------------------------------------
    # RELATIONSHIP
    # --------------------------------------------------------
    #
    # Esto NO es una columna de la tabla.
    #
    # Permite acceder directamente al objeto Team:
    #
    # hero.team
    #
    team: Team | None = Relationship(
        back_populates="heroes"
    )
